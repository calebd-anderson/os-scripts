LDAP - Dovecot

yum install -y openldap*
create binddn user in AD
ldapsearch -h ad.metro-ccdc.lab -b ou=mailusers,dc=metro-ccdc,dc=lab -LLL -x -W -D cn=binddn,cn=users,dc=metro-ccdc,dc=lab

create local vmail user

groupadd -g 10000 vmail useradd -u 10000 -g 10000 -s /bin/false vmail

or create system user

groupadd -g 5000 -r vmail adduser --system --home /srv/vmail --uid 5000 --gid 5000 vmail

create virtual mail directory

mkdir /etc/dovecot/users chown vmail:vmail /etc/dovecot/users/ chmod 700 /etc/dovecot/users/

ldap/ad config

vi /etc/dovecot/dovecot-ldap.conf.ext

        driver = ldap

LDAP

    ldapaliases_server_host = ad.clownfish.com
    ldapaliases_search_base = ou="mail users",dc=ad,dc=clownfish,dc=com
    ldapaliases_query_filter = userPrincipalName=%s
    ldapaliases_result_attribute = userPrincipalName
    ldapaliases_bind = yes
    ldapaliases_bind_dn = cn=binddn,ou="mail users",dc=ad,dc=clownfish,dc=com
    ldapaliases_bind_pw = changeME!

commands

config

all Postfix parameters and their values

postconf

value of a parameter

postconf [parameter]

all parameters explicitly set in main.cf

postconf -n

default values

postconf -d

defualt value of a given parameter

postconf -d alias_maps

queue

all mail in queue

postqueue -p

flush the queue (try to send)

postqueue -f

Delete an email from the queue (ids can be found in maillog)

postsuper -d [id]

delete all from queue

postsuper -d ALL

read message in queue

postcat -q [messageid]

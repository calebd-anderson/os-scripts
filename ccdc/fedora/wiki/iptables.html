iptables

reference

Disable firewalld:

systemctl stop firewalld systemctl disable firewalld systemctl mask firewalld

Reset (nuke the rules)

iptables -F iptables -X iptables -P INPUT ACCEPT iptables -P OUTPUT ACCEPT iptables -P FORWARD ACCEPT

Config Files

/etc/sysconfig/iptables-config
/proc/net/[...]
/etc/protocols

Aggregate (CCDC)

iptables -A INPUT -p tcp -m multiport --dports 25,80,110,143 -m state --state NEW,ESTABLISHED -j ACCEPT iptables -A OUTPUT -p tcp -m multiport --sports 25,80,110,143 -m state --state NEW,ESTABLISHED -j ACCEPT iptables -I INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT iptables -A OUTPUT -p udp -m multiport --dports 53,123,389 -m state --state NEW,ESTABLISHED -j ACCEPT iptables -A OUTPUT -p tcp -m multiport --dports 389,443 -m state --state NEW,ESTABLISHED -j ACCEPT iptables -A OUTPUT -p icmp --icmp-type echo-request -j ACCEPT iptables -A INPUT -i lo -j ACCEPT iptables -A OUTPUT -o lo -j ACCEPT

Allow SSH - 22/tcp

iptables -A INPUT -p tcp --dport 22 -j ACCEPT iptables -A OUTPUT -p tcp --sport 22 -j ACCEPT

Allow SMTP (postfix) - 25/tcp (?)

iptables -A INPUT -p tcp --dport 25 -j ACCEPT iptables -A OUTPUT -p tcp --sport 25 -j ACCEPT

Allow DNS - 53/udp

iptables -A INPUT -p udp --sport 53 -j ACCEPT iptables -A OUTPUT -p udp --dport 53 -j ACCEPT

Allow HTTP (apache) - 80/tcp

iptables -A INPUT -p tcp --dport 80 -j ACCEPT iptables -A OUTPUT -p tcp --sport 80 -j ACCEPT

Allow POP3 (dovecot) - 110/tcp

iptables -A INPUT -p tcp --dport 110 -j ACCEPT iptables -A OUTPUT -p tcp --sport 110 -j ACCEPT

Allow NTP - 123/tcp

iptables -A INPUT -p udp --sport 123 -j ACCEPT iptables -A OUTPUT -p udp --dport 123 -j ACCEPT

Allow IMAP - 143/tcp

iptables -A INPUT -p tcp --dport 143 -j ACCEPT iptables -A OUTPUT -p tcp --sport 143 -j ACCEPT

ALLOW LDAP - 389/tcp

iptables -A INPUT -p udp --sport 389 -j ACCEPT iptables -A OUTPUT -p udp --dport 389 -j ACCEPT iptables -A INPUT -p tcp --sport 389 -j ACCEPT iptables -A OUTPUT -p tcp --dport 389 -j ACCEPT

Allow HTTPS (wget) - 443/tcp

iptables -A INPUT -p tcp --sport 443 -j ACCEPT iptables -A OUTPUT -p tcp --dport 443 -j ACCEPT

Allow Loopback

iptables -A INPUT -i lo -j ACCEPT iptables -A OUTPUT -o lo -j ACCEPT

Allow established and related back in

iptables -I INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT iptables -I INPUT -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT iptables -I OUTPUT -m conntrack --ctstate ESTABLISHED -j ACCEPT reference

Allow ICMP Out

iptables -A OUTPUT -p icmp --icmp-type echo-request -j ACCEPT

Reject IP

iptables -A INPUT -s <IPADDR> -j REJECT

Show Line Numbers

iptables -L --line-numbers

Delete by Line Number

iptables -D INPUT <n>

Replace by Line Number

iptables -R <chain> <rulenum> <new rule def>

Port Range

iptables -A tableName -p protocol --match multiport --dports portRange1:PortRange2 -j ACCEPT

List Rules

iptables -S iptables -L -v -n

ldap_virtual_recipients.cf

vi /etc/postfix/ldap/ldap_virtual_recipients.cf 
        server_host = ldap://ad.metro-ccdc.lab
        server_port = 389
        search_base = cn=Users,dc=metro-ccdc,dc=lab
        version = 3
        bind = yes
        bind_dn = cn=caleb,cn=Users,dc=metro-ccdc,dc=lab
        bind_pw = ***REMOVED***
        query_filter = cn=%s
        result_attribute = cn
postmap -q 'caleb' ldap:/etc/postfix/ldap/ldap_virtual_recipients.cf

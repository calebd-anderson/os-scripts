ad_virtual_recipients.cf

vi /etc/postfix/ldap/ldap_virtual_recipients.cf 
        server_host = ldap://ad.metro-ccdc.lab
        server_port = 389
        search_base = ou=MailUsers,dc=metro-ccdc,dc=lab
        version = 3
        bind = yes
        bind_dn = cn=vmail,cn=Users,dc=metro-ccdc,dc=lab
        bind_pw = changeME!
        query_filter = mail=%s
        result_attribute = mail

useful commands

config

all Postfix parameters and their values

postconf

give value of a parameter

postconf smtpd_tls_security_level

all paramaters explicitly set in main.cf

postconf -n

give default values

postconf -d

defualt value of a given parameter

postconf -d alias_maps

queue

all mail in queue

postqueue -p

flush the queue (try to send)

postqueue -f

Delete an email from the queue (ids can be found in maillog)

postsuper -d [id]

delete all from queue

postsuper -d ALL

display message in queue

postcat -q [messageid]

logging

egrep '(reject|warning|error|fatal|panic):' /var/log/maillog

important paramaters

postconf local_recipient_maps

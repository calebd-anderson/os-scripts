LDAP - Dovecot

yum install -y openldap*
create binddn user in AD
ldapsearch -h ad.metro-ccdc.lab -b ou=mailusers,dc=metro-ccdc,dc=lab -LLL -x -W -D cn=binddn,cn=users,dc=metro-ccdc,dc=lab

create local vmail user

groupadd -g 10000 vmail useradd -u 10000 -g 10000 -s /bin/false vmail chown vmail:vmail /etc/dovecot/users/ chmod 700 /etc/dovecot/users/

config

vi /etc/dovecot/dovecot-ldap.conf.ext 
    hosts           = ad.metro-ccdc.lab:389
    ldap_version    = 3
    auth_bind       = yes
    dn              = binddn
    dnpass          = changeME!
    #base            = cn=users,dc=metro-ccdc,dc=lab
    base            = ou=mailusers,dc=metro-ccdc,dc=lab
    #scope           = subtree
    #deref           = never

    #user_filter     = (&(objectClass=Person)(sAMAccountName=%n))
    #pass_filter     = (&(objectClass=Person)(sAMAccountName=%n))

    user_filter     = (&(objectClass=Person)(userPrincipalName=%u))
    pass_filter     = (&(objectClass=Person)(userPrincipalName=%u))

    pass_attrs      = userPassword=password

symlink so passdb/userdb

ln -s /etc/dovecot/dovecot-ldap.conf.ext /etc/dovecot/dovecot-ldap-userdb.conf.ext vi +21 /etc/dovecot/conf.d/auth-ldap.conf.ext 
    userdb {
        driver = ldap
        + args = /etc/dovecot/dovecot-ldap-userdb.conf.ext
        - args = /etc/dovecot/dovecot-ldap.conf.ext
    }
